Todos os endpoints exigem autenticação via Bearer Token no cabeçalho `Authorization`.

---

## Registrar ocorrência de entrega

Insere no sistema ocorrências para uma entrega

**`POST /tracking/ocorrencias`**

### Corpo da requisição

<ParamField body="auth" type="object">
  Opcional. Só deve ser enviado caso o cabeçalho Authorization não for utilizado.

  <Expandable title="Propriedades de auth">
    <ParamField body="usuario" type="string">
  
    </ParamField>
    <ParamField body="senha" type="string">
  
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="documentos" type="array of objects">
  

  <Expandable title="Cada item de documentos">
    <ParamField body="cliente" type="string" required>
      CNPJ do cliente. Quando o tipo for manifesto ou despacho, CNPJ não é obrigatório.
    </ParamField>
    <ParamField body="tipo" type="string" required>
      Valores aceitos: `PEDIDO`, `NF`, `NFE`, `CTE`, `MINUTA`, `COLETA`, `MANIFESTO`, `DESPACHO`.
    </ParamField>
    <ParamField body="tipo_op" type="string">
      Valores aceitos: `MINUTA`, `COLETA`.
    </ParamField>
    <ParamField body="numero" type="integer">
      Campo obrigatório quando o tipo igual a NF ou PEDIDO
    </ParamField>
    <ParamField body="serie" type="string">
      Campo obrigatório quando o tipo igual a NF
    </ParamField>
    <ParamField body="chave" type="string">
      Campo obrigatório quando o tipo igual a NFE ou CTE
    </ParamField>
    <ParamField body="minuta" type="integer">
      Campo obrigatório quando o tipo igual a MINUTA
    </ParamField>
    <ParamField body="coleta" type="integer">
      Campo obrigatório quando o tipo igual a COLETA
    </ParamField>
    <ParamField body="manifesto" type="integer">
      Campo obrigatório quando o tipo igual a MANIFESTO
    </ParamField>
    <ParamField body="awb" type="integer">
      Campo obrigatório quando o tipo igual a DESPACHO
    </ParamField>
    <ParamField body="eventos" type="array of objects" required>
      Lista de eventos

      <Expandable title="Cada item de eventos">
        <ParamField body="codigo" type="integer" required>
          Código da ocorrência no sistema Brudam
        </ParamField>
        <ParamField body="data" type="string" required>
          Data e hora da ocorrência
        </ParamField>
        <ParamField body="obs" type="string">
          Observação
        </ParamField>
        <ParamField body="recebedor" type="object">
          Caso enviado o campo recebedor, os campos 'nome' e 'documento' são obrigatórios

          <Expandable title="Propriedades de recebedor">
            <ParamField body="nome" type="string" required>
              Nome do recebedor
            </ParamField>
            <ParamField body="documento" type="string" required>
              Documento do recebedor
            </ParamField>
            <ParamField body="grau" type="string">
              Grau de parentesco
            </ParamField>
          </Expandable>
        </ParamField>
        <ParamField body="agendamento" type="object">
  

          <Expandable title="Propriedades de agendamento">
            <ParamField body="data" type="string">
              Data do agendamento
            </ParamField>
            <ParamField body="hIni" type="string">
              Hora inicial
            </ParamField>
            <ParamField body="hFim" type="string">
              Hora final
            </ParamField>
          </Expandable>
        </ParamField>
        <ParamField body="localizacao" type="object">
          Coordenadas geográficas do evento (opcional), latitude e longitude.

          <Expandable title="Propriedades de localizacao">
            <ParamField body="latitude" type="number">
  
            </ParamField>
            <ParamField body="longitude" type="number">
  
            </ParamField>
          </Expandable>
        </ParamField>
      </Expandable>
    </ParamField>
    <ParamField body="anexos" type="array of objects">
      Lista de anexos

      <Expandable title="Cada item de anexos">
        <ParamField body="arquivo" type="object">
  

          <Expandable title="Propriedades de arquivo">
            <ParamField body="nome" type="string" required>
              Nome ou URL do arquivo
            </ParamField>
            <ParamField body="dados" type="string">
              Conteúdo do arquivo em *Base64* quando **Nome** não for uma URL
            </ParamField>
          </Expandable>
        </ParamField>
      </Expandable>
    </ParamField>
  </Expandable>
</ParamField>

### Exemplo de requisição

<CodeGroup>

```bash cURL
curl --request POST \
  --url "https://{cliente}.brudam.com.br/api/v1/tracking/ocorrencias" \
  --header "Authorization: Bearer {seu_token}" \
  --header "Content-Type: application/json" \
  --data '{
  "auth": {
    "usuario": "string",
    "senha": "string"
  },
  "documentos": [
    {
      "cliente": "string",
      "tipo": "PEDIDO",
      "tipo_op": "MINUTA",
      "numero": 0
    }
  ]
}'
```

</CodeGroup>

### Respostas

**`200`** — Evento(s) registrado(s) com sucesso!

```json
{
  "message": "OK",
  "status": 1,
  "data": [
    {
      "cliente": "02801696000180",
      "tipo": "MINUTA",
      "minuta": "143636",
      "eventos": [
        {
          "codigo": 150,
          "data": "2019-06-04 13:21:25",
          "obs": null
        }
      ],
      "numero": null,
      "serie": null,
      "chave": null,
      "anexos": null,
      "status": "1",
      "message": "Evento(s) registrado(s) com sucesso!",
      "messages": [
        {
          "status": 0,
          "codigo": 26,
          "message": "Ocorrência já foi inserida nesta coleta!"
        },
        {
          "status": 1,
          "codigo": 91,
          "message": "Ocorrência inserida com sucesso!"
        }
      ]
    }
  ]
}
```

```json
{
  "message": "Erro nos dados enviados",
  "status": 0,
  "data": [
    "O campo Senha deve conter um valor de acordo com a E.R. ^[0-9A-Fa-f]{64}.",
    "O campo Senha deve ter exatamente 64 caracteres"
  ]
}
```

**`401`** — Se o usuário não possui permissão para acessar a URL informada ou o TOKEN não for informado.

```json
{
  "message": "Acesso não autorizado",
  "status": 0
}
```

**`403`** — Se os dados de usuário estão incorretos.

```json
{
  "message": "Acesso negado!",
  "status": 0
}
```

**`404`** — Se a URL informada não for encontrada

```json
{
  "message": "Nenhum resultado encontrado!",
  "status": 0
}
```

**`405`** — Se a URL chamada não suporta o método utilizado

```json
{
  "message": "Método não permitido!",
  "status": 0
```
